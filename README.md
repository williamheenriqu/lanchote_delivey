// Cardápio e simulador do Sabor na Porta
const comidas=[{id:"hamburguer",nome:"Hambúrguer artesanal",preco:28,emoji:"🍔",**https://share.google/jLRn1ahNMO6spVEC8**,,{id:"queijo",nome:"Hambúrguer com queijo",preco:30,emoji:"🧀",img:"https://images.unsplash.com/photo-https://share.google/YsPdFYKPy6JziefN7;
const bebidas=[{id:"coca",nome:"Coca-Cola",preco:7,emoji:"🥤",img:"https://images.unsplash.com/photo-1554866585-cd94860890b7?w=600"},{id:"uva",nome:"Suco de uva",preco:8,emoji:"🍇",img:"https://images.unsplash.com/photo-1534353473418-4cfa6c56fd38?w=600"}];
const TAXA_MOTOBOY=7,VALOR_DESCONTO=100,PERCENTUAL=0.05;
const moeda=v=>v.toLocaleString("pt-BR",{style:"currency",currency:"BRL"});
function card(p){return `<article class="produto"><img src="${p.img}" alt="${p.nome}" loading="lazy"><h3>${p.emoji} ${p.nome}</h3><p class="price">${moeda(p.preco)}</p></article>`}
function exibirCardapio(){const c=document.getElementById("menu-comidas"),b=document.getElementById("menu-bebidas");if(c)c.innerHTML=comidas.map(card).join("");if(b)b.innerHTML=bebidas.map(card).join("")}
function linhaPedido(p){return `<div class="line"><span>${p.emoji} ${p.nome}<br><b>${moeda(p.preco)}</b></span><label>Qtd. <input type="number" min="0" max="99" value="0" data-preco="${p.preco}" aria-label="Quantidade de ${p.nome}"></label></div>`}
function iniciarPedido(){const c=document.getElementById("pedido-comidas"),b=document.getElementById("pedido-bebidas");if(!c||!b)return;c.innerHTML=comidas.map(linhaPedido).join("");b.innerHTML=bebidas.map(linhaPedido).join("");document.querySelectorAll("[data-preco]").forEach(i=>i.addEventListener("input",calcularTotal));document.getElementById("finalizar-pedido").addEventListener("click",()=>{const q=[...document.querySelectorAll("[data-preco]")].reduce((s,i)=>s+(Number(i.value)||0),0);document.getElementById("mensagem-pedido").textContent=q?"Pedido calculado! Esta é uma simulação: não enviamos pedidos reais.":"Escolha pelo menos um produto.";});calcularTotal()}
function calcularTotal(){const campos=[...document.querySelectorAll("[data-preco]")];if(!campos.length)return;const subtotal=campos.reduce((s,i)=>s+Math.max(0,Number(i.value)||0)*Number(i.dataset.preco),0),desconto=subtotal>=VALOR_DESCONTO?subtotal*PERCENTUAL:0,total=subtotal+TAXA_MOTOBOY-desconto;document.getElementById("subtotal").textContent=moeda(subtotal);document.getElementById("entrega").textContent=moeda(TAXA_MOTOBOY);document.getElementById("desconto").textContent=moeda(desconto);document.getElementById("total").textContent=moeda(total);document.getElementById("mensagem-desconto").textContent=desconto?"Parabéns! Você ganhou 5% de desconto nos produtos.":"Faltam "+moeda(Math.max(0,VALOR_DESCONTO-subtotal))+" em produtos para ganhar 5% de desconto."}
function validarCadastro(){const f=document.getElementById("form-cadastro");if(!f)return;f.addEventListener("submit",e=>{e.preventDefault();const idade=Number(f.elements.idade.value);if(idade<1||idade>120){document.getElementById("mensagem-cadastro").textContent="Informe uma idade válida.";return}document.getElementById("mensagem-cadastro").textContent="Cadastro preenchido! (Os dados não são armazenados.)";f.reset()})}
document.addEventListener("DOMContentLoaded",()=>{exibirCardapio();iniciarPedido();validarCadastro()});
